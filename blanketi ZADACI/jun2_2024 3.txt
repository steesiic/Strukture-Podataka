Node* minEvenDiff()
{
    if(root==nullptr || root->left==nullptr || root->right==nullptr)
        return nullptr;
        
        Node* result=nullptr;
        int minDiff=0;
        findMinEvenDiff(root, minDiff, result);
        return result;
}

int findMinEvenDiff(Node* node, int& minDiff, Node* result)
{
    if(node==nullptr)
	return 0;
	int left= findMinEvenDiff(node->left,  minDiff, result);
	int right= findMinEvenDiff(node->right, minDiff, result);
	
	if(abs(left-right)<minDiff)
	{
	    minDiff= abs(left-right);
	    result= node;
	}
	if(node->info %2 == 0)
	{
	    return left+right+node->info;
	}
	else   
	    return left+right;
	
}
