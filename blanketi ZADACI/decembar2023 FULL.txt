{

    lastNodePrev->next = pomB;
    Node* pom = pomA->next;
    pomA->next= pomB->next;
    pomB->next= pom;

        head=pomB;

// Struktura za čuvanje broja pobeda i broja čvorova koji zadovoljavaju uslov
struct Result {
    int winsDom;
    int winsGos;
    int count;
};

Result count(LinkedNode* node)
{
    if(node == nullptr)
        return {0, 0, 0};

    Result left = count(node->left);
    Result right = count(node->right);

    // Ukupan broj pobeda u podstablu
    int totalDom = left.winsDom + right.winsDom + (node->info->dom > node->info->gos ? 1 : 0);
    int totalGos = left.winsGos + right.winsGos + (node->info->gos > node->info->dom ? 1 : 0);

    // Broj čvorova koji zadovoljavaju uslov
    int totalCount = left.count + right.count;
    if(totalDom > totalGos)
        totalCount += 1;

    return {totalDom, totalGos, totalCount};
}

int LongestSimpleCycle()
{
    int max=0;
    LinkedNodeInt* ptr = start;
    while(ptr!=nullptr)
    {
        int i= dfsHelper(ptr);
        if(i>max)
        {
            max=i;

        ptr=ptr->next;
    }
    return max;
}
