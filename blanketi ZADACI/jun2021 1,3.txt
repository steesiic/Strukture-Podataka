int countSl(Node* root)
{
	int count=0;
	
	if(root==nullptr) return 0;

	
	int zbirLevog = sum(root->left);
	int zbirDesnog = sum(root->right);
	
	if(zbirLevog>zbirDesnog)
	{
		count++;
	}
	count+=countSl(root->left);
	count+=countSl(root->right);
	
	return count;
}

int sum(Node* node)
{
	if(node==nullptr)return 0;
	
	return node->info+sum(node->left)+sum(node->right);
