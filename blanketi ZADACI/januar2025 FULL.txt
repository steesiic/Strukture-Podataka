void partitionList(Node** head, int pivot)
{
    if(head==nullptr || *head==nullptr) 
        return;
    
        Node* curr = *head;
        Node* prevCurr = nullptr;
        
        while(curr != nullptr)
        {
            Node* tmpCurr = curr;
            Node* tmpPrevCurr = prevCurr;
            
            Node* veci = curr;
            Node* prevVeci = prevCurr;
            while(veci != nullptr && veci->info >=pivot)
            {
                prevVeci = veci;
                veci=veci->next;
            }
            if(veci !=nullptr && veci->info<pivot)
            {
                prevVeci->next = veci->next;
                if(tmpPrevCurr==nullptr)
                {
                    veci->next = *head;
                    *head = veci;
                }
                
                else
                {
                    tmpPrevCurr->next = veci;
                    veci->next = curr;
                }
            }
            prevCurr = curr;
            curr=  curr->next;
        }
}

minConnectingFlights(Node* start, Node* goal1, Node* goal2, int& diff)
{
    Queue *q = new Queue(NodeNum);
    q->enqueue(start);
    start->status = 1;
    start->prev = nullptr; //prev kao prethodnik(predecessor)
    
    while(!q.isEmpty())
    {
            Node* curr = q->dequeue();
            Node* neighbor = curr->adj;
            while(neighbor != nullptr)
            {
                if(neighbor->status == 0)
                {
                    neighbor->status=1;
                    neighbor->prev = curr;
                    q->enqueue(neighbor);
                }
                neighbor=neighbor-> nextAdj;
            }
    }
    // Brojanje presedanja do goal1
    int steps1 = 0;
    Node* tmp = goal1;
    while(tmp->prev != nullptr)
    {
        tmp = tmp->prev;
        steps1++;
    }

    // Brojanje presedanja do goal2
    int steps2 = 0;
    tmp = goal2;
    while(tmp->prev != nullptr)
    {
        tmp = tmp->prev;
        steps2++;
    }

    // Razlika u broju presedanja
    diff = (steps1 > steps2) ? (steps1 - steps2) : (steps2 - steps1);

    // Vraća čvor do kojeg je manje presedanja
    return (steps1 <= steps2) ? goal1 : goal2;
    
    
    
}
