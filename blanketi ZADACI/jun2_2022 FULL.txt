Llist* SplitOrMove()
{
  LList* novaLista = new LList();
novaLista->head=nullptr;
if(isEmpty()) return novaLista;

LListNode* currPrev = nullptr;
LListNode* curr  = head;

while (curr!=nullptr)
{
    LListNode tmpPrev=curr;
    LListNode tmp = curr->next;
	while(tmp !=nullptr && tmp->info!=curr->info)
	{
		tmpPrev=tmp;
		tmp=tmp->next;	
	}
		
	LListNode* forMove;
	if(tmp==nullptr)
	{
		forMove= curr;
		if(currPrev==nullptr)
		{
			head=head->next;
			curr=head;
		}	
		else
		{
			currPrev->next= curr->next;
			curr=curr->next;
		}

	}
	else
	{
		forMove=tmp;
		tmpPrev->next = tmp->next;
		currPrev=curr;
		curr=curr->next;	
	}
	forMove->next = novaLista->head;
	novaLista->head=forMove;


}
return novaLista;
}

int brojCvorova(int keyMin, valueMin)
{
    if(root==nullptr)
    {
        return 0;
    }
    bool predak = false;
    return brojCvorovaCount(root,keyMin,valueMin,predak);
}

int brojCvorovaCount(BSTNode* node, int keyMin,int valueMin,bool predak)
{
    if(node==nullptr)
        return 0;
    
    bool uslov1= node->key > keyMin;
    bool uslov2 = predak;
    int count = 0;
    if(uslov1 && uslov2)
        ++count;
        
    bool noviPredak = predak || (node->value > valueMin);    
    
    count+= brojCvorovaCount(node->left, keyMin,valueMin, noviPredak);
    count+= brojCvorovaCount(node->right, keyMin,valueMin, noviPredak);
    return count;
    
}
