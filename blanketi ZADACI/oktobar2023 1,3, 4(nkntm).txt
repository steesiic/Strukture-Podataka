void rearrange(int v1,int v2, int k)
{
    if(head==nullptr)
        return;
    int j=0;
    Node* ptrK = head;
    while(j<k)
    {
        ptrK=ptrK->next;
        j++;    
    }
    if(ptrK == nullptr) 
        return;
            
    Node* preK=head;
    Node* preKPrev=nullptr;
    Node* posleK=ptrK->next;
    Node* posleKPrev = ptrK;
    
    while(preK!=ptrK)
    {
       while(preK->info >= v1)
       {
           preKPrev=preK;
           preK=preK->next;
       }
       if(preK==ptrK) break;
       while(posleK->info <= v2)
       {
           posleKPrev=posleK;
           posleK=posleK->next;
       }
       Node *tmp1,*tmp2,*tmp3,*tmp4;
       tmp1=preKPrev;
       tmp2=preK;
       tmp3=posleKPrev;
       tmp4=posleK;
       Swap(tmp1,tmp2,tmp3,tmp4);
       
       preKPrev=preK;
       preK=preK->next;
       posleKPrev=posleK;
       posleK=posleK->next;
    }    
}

BSTNode* findDeepestLeaf()
{
    if(root==nullptr)
        return nullptr;
    if(root->left==nullptr && root->right==nullptr)
        return root;
    
    BSTNode* result = nullptr;
    int level=0;
    int maxLevel=0;
    
    DeepestLeaf(root,result,level,maxLevel);
    return result;
}

void DeepestLeaf(BSTNode* node, BSTNode*& result,int level, int& maxLevel)
{
    if(node==nullptr)
        return;
        
        if(node->left!=nullptr)
            findDeepestLeaf(node->left,result,1+level,maxLevel);
        
        if(node->right!=nullptr)
            findDeepestLeaf(node->right,result,1+level,maxLevel);
        
        if(level>maxLevel)
        {
            maxLevel=level;
            result=node;
        }
        return;
}

int bfsHelper(LinkedNodeInt* ptr, LinkedNodeInt* ptr2)
{
    if(ptr==nullptr)
        return INT_MAX;
    Queue q = Queue(nodeNum);
    q.enqueue(ptr);
    ptr->status=1;
    int i=1;
    while(!q.isEmpty())
    {
        LinkedNodeInt* current = q.pop();
        LinkedNodeEdge* edge = current->adj;
        while(edge!=nullptr)
        {   
            if(edge->dest->status==0)
            {
                q.enqueue(edge->dest);
                edge->dest->status=1;
            }
            if(edge->dest == ptr2)
            {
                return i+1;
            }
            edge=edge->link;
        }
        i++;
        ptr->status=2;
    }
    return i;
}

bool canReach(int A, int B, int noNodes)

    int count = bfsHelper(a, b);
    if(count > 0 && count <= noNodes)
        return true;
        
    return false;
}


int bfsHelper(LinkedNodeInt* ptr, LinkedNodeInt* ptr2)
{
    if(ptr == nullptr)
        return 0;

    Queue q(nodeNum);
    q.enqueue(ptr);
    ptr->status = 1;

    int level = 1; // broj čvorova u putu
    
    while(!q.isEmpty())
    {
        LinkedNodeInt* current = q.pop();

        LinkedNodeEdge* edge = current->adj;
        while(edge != nullptr)
        {
            if(edge->dest->status == 0)
            {
                edge->dest->status = 1;
                q.enqueue(edge->dest);
            }

            if(edge->dest == ptr2)
            {
                return level + 1; // uključuje i krajnji čvor
            }

            edge = edge->link;
        }

        current->status = 2;
    }

    return 0; // nema puta
}
