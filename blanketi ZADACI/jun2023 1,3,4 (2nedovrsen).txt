		}

		if(ok && tmpSublist==nullptr)

void LList::deleteFrom(LListNode* p, int brEl)
{
	int br = 0;
	LListNode* curr;
	LListNode* next;
	if (p == nullptr)
		curr = head;
	
	else 
		curr = p->next;

	while(br<brEl)
	{
		next = curr->next;
		delete curr;
		curr = next;
		++br;
	}
	if (p == nullptr)
		head = curr;
	else
		p->next = curr;
}

int findNodeWithMaxHeight(Node* node, Node** returnNode, int& maxDiff)
{
    if(node==nullptr)
        return 0;
    
     int leftHeight = findNodeWithMaxHeight(node->left, returnNode,maxDiff);
     int rightHeight= findNodeWithMaxHeight(node->right,returnNode,maxDiff);
     
     if(abs(leftHeight-rightHeight)>maxDiff)
     {
         maxDiff=abs(leftHeight-rightHeight);
         *returnNode= node;
     }
     int a=rightHeight;
     if(leftHeight>rightHeight){a=leftHeight;}
     return 1+a;
}

void connectMaxAndMin()
{
    LinkedNodeInt* ptr = start;
    LinkedNodeInt* fromMax=nullptr;
    LinkedNodeInt* fromMin=nullptr;
    SetAllNodeStatuses(0);
    int max=0;
    int min= INT_MAX;
    
    while(ptr!=nullptr)
    {
        if(ptr->status==0)
        {
            int rez=dfsHelper(ptr);
        }
        if(rez>max)
        {
            max=rez;
            fromMax=ptr;
        }
        if(rez<min)
        {
            min=rez;
            fromMin=ptr;
        }
        ptr=ptr->next;
    }
    if (fromMax != nullptr && fromMin != nullptr && fromMax != fromMin)
    {
        InsertEdge(fromMax, fromMin);
    }
    
}

int dfsHelper(LinkedNodeInt* ptr)
{
   StackAsArray stack(nodeNum);
   stack.push(ptr);
   ptr->status=1;
   int obradjen =0;
   while(!stack.isEmpty())
   {
       LinkedNodeInt* current = stack.pop();
       LinkedNodeEdge* edge = current->adj;
       while(edge!=nullptr)
       {
           if(edge->dest->status == 0)
            {
                stack.push(edge->dest);
                edge->dest->status = 1;
            }
            edge = edge->link;
       }
       current->status=2;
       obradjen++;
   }
   return obradjen;
}
