void selectSort(DLLNode* head)
{
	if(head==nullptr || //head==tail head->next==nullptr)
	return;
	
	DLLNode* i=head;
	DLLNode* j;
	DllNode* min;
	while(i->next !=nullptr)
	{
		 min=i;
		j=i->next;

		while(j!=nullptr)
		{
			if(j->info < min->info)
				min=j;
			j=j->next;
		}	
		
		if(min!=i)
		{  
			DLLNode* prevI = i->prev;
    			DLLNode* nextI = i->next;
    			DLLNode* prevM = min->prev;
    			DLLNode* nextM = min->next;
			if(i->next==min)
			{
				prevI->next=min;
				min->prev=prevI;
				min->next=i;
				i->prev=min;
				i->next=nextM;	
				nextM->prev=i;	

			}
			else
			{    
				prevI->next=min;
				nextI->prev=min;
				prevM->next=i;
				nextM->prev=i;

				min->prev=prevI;
				min->next=nextI;
				i->prev=prevM;
				i->next=nextM;

			}	

		}
		if(i==head)
			head=min;

		i=min->next;
	}
	
}

BSTNode* shallowestLeaf(BSTNode* root)
{
	if(root==nullptr)
	{
		return nullptr;
	}
	if(root->left == nullptr && root->right==nullptr)
        	return root;

	int minDepth = INT_MAX;
	BSTNode* result = nullptr;
	findShallowestLeaf(root,0,minDepth,result);
	return result;
}

void findShallowestLeaf(BSTNode* node, int depth, int& minDepth, BSTNode*& result)
{
	if(node==nullptr) result=nullptr;
	
	if(node->left == nullptr && node->right == nullptr)
	{
		if(depth<minDepth)
		{
			minDepth=depth;
			result=node;
		}
		return node;
	}

	findShallowestLeaf(node->left, depth+1, minDepth, result);
	findShallowestLeaf(node->right, depth+1, minDepth, result);
	return result;
}

            LinkedEdgeInt* edg=ptr->adj;
