{

    unsigned int h=0;
    int i=0;

    {
        h=h*101+ime[i];
        i++;
    }
    i=0;
    while(prezime[i]!='\0')
    {
         h=h*101+prezime[i];
        i++;
    }
    i=0;
    while(jmbg[i]!='\0')
    {
        h=h*101+jmbg[i];
        i++;
        
    }
    return(h*a)>>(32-14);
}

bool isCyclic()
{
    Node* ptr=start;
    SetAllNodeStatuses(0); //i ova f-ja mora da se napise, napisana na drugim blank.
    
    while(ptr!=nullptr)
    {
        if(ptr->status==0)
        {
            if(dfs(ptr))
            {
                return true;
            }
            ptr=ptr->next;
        }
    }
    return false;
}

bool dfs(Node* ptr)
{
    ptr->status=1;
    Edge edg=ptr->adj;
    while(edg!=nullptr)
    {
        if(edg->dest->status==1)
            return true;
        else if((edg->dest->status==0)&& dfs(edg->dest))
            return true;
        edg=edg->link;
    }
    ptr->status=2;
    return false;
}
