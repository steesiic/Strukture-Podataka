    {
         Node* tmp = curr->next;

       {

           {
               tmpPrev->next = tmp->next;

               curr=curr->next;
               tmp=tmpPrev->next;

           }
           else
           {
               tmpPrev=tmp;
               tmp=tmp->next;
           }

    }
 
}
